    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // --- SCROLL EFFECT ON HEADER ---
            const header = document.getElementById('main-header');
            const brandText = document.getElementById('brand-text');
            const desktopLinks = document.querySelectorAll('.desktop-link');
            const getQuoteBtn = document.getElementById('get-quote-btn');
            const menuIcon = document.getElementById('menu-icon');
            const mobileMenuBtn = document.getElementById('mobile-menu-btn');
            const mobileNavOverlay = document.getElementById('mobile-nav-overlay');
            const closeIcon = document.getElementById('close-icon');
            const mobileNavLinks = document.querySelectorAll('.mobile-nav-link');
            
            let isMenuOpen = false;

            const handleScroll = () => {
                const isScrolled = window.scrollY > 20;

                if (isScrolled) {
                    header.classList.remove('bg-transparent', 'border-transparent', 'py-5');
                    header.classList.add('bg-white/90', 'backdrop-blur-md', 'border-stone-200', 'py-3', 'shadow-sm');

                    if(!isMenuOpen) {
                        brandText.classList.remove('lg:text-white');
                        brandText.classList.add('text-stone-900');
                    }

                    desktopLinks.forEach(link => {
                        link.classList.remove('text-white/90');
                        link.classList.add('text-stone-600');
                    });

                    getQuoteBtn.classList.remove('bg-white', 'text-stone-900', 'hover:bg-stone-100');
                    getQuoteBtn.classList.add('bg-stone-900', 'text-white', 'hover:bg-stone-800');

                    if(!isMenuOpen) {
                        menuIcon.classList.remove('md:text-white');
                        menuIcon.classList.add('text-stone-900');
                    }
                } else {
                    header.classList.add('bg-transparent', 'border-transparent', 'py-5');
                    header.classList.remove('bg-white/90', 'backdrop-blur-md', 'border-stone-200', 'py-3', 'shadow-sm');

                    if(!isMenuOpen) {
                        brandText.classList.add('lg:text-white');
                        // Ensure it stays text-stone-900 on mobile size, but allow lg:text-white to work
                    }

                    desktopLinks.forEach(link => {
                        link.classList.add('text-white/90');
                        link.classList.remove('text-stone-600');
                    });

                    getQuoteBtn.classList.add('bg-white', 'text-stone-900', 'hover:bg-stone-100');
                    getQuoteBtn.classList.remove('bg-stone-900', 'text-white', 'hover:bg-stone-800');

                    if(!isMenuOpen) {
                        menuIcon.classList.add('md:text-white');
                        // Ensure text-stone-900 stays for mobile size via Tailwind classes
                    }
                }
            };

            window.addEventListener('scroll', handleScroll);
            handleScroll(); // Initialize on load

            // --- MOBILE MENU TOGGLE ---
            const toggleMenu = () => {
                isMenuOpen = !isMenuOpen;
                if (isMenuOpen) {
                    mobileNavOverlay.classList.remove('translate-x-full');
                    mobileNavOverlay.classList.add('translate-x-0');
                    
                    menuIcon.classList.add('hidden');
                    closeIcon.classList.remove('hidden');
                    
                    // Force brand text to be dark when menu is open
                    brandText.classList.remove('lg:text-white');
                    brandText.classList.add('text-stone-900');
                } else {
                    mobileNavOverlay.classList.add('translate-x-full');
                    mobileNavOverlay.classList.remove('translate-x-0');
                    
                    closeIcon.classList.add('hidden');
                    menuIcon.classList.remove('hidden');
                    
                    // Re-evaluate scroll state for text colors
                    handleScroll();
                }
            };

            mobileMenuBtn.addEventListener('click', toggleMenu);
            
            // Close menu when clicking a link
            mobileNavLinks.forEach(link => {
                link.addEventListener('click', () => {
                    if(isMenuOpen) toggleMenu();
                });
            });

            // --- FORM SUBMISSION TO WHATSAPP ---
            const form = document.getElementById('quote-form');
            const successMessage = document.getElementById('success-message');
            const submitBtn = document.getElementById('submit-btn');
            const btnText = document.getElementById('btn-text');
            const spinner = document.getElementById('btn-spinner');
            const resetBtn = document.getElementById('reset-form-btn');

            form.addEventListener('submit', (e) => {
                e.preventDefault();
                
                // Show submitting state
                submitBtn.disabled = true;
                btnText.textContent = 'Sending...';
                spinner.classList.remove('hidden');

                const formData = new FormData(form);
                const name = formData.get('name');
                const phone = formData.get('phone');
                const service = formData.get('service');
                
                const serviceNames = {
                    upholstery: 'Upholstery',
                    curtains: 'Custom Curtains',
                    wood: 'Wood Refurbishing',
                    other: 'Other'
                };
                const selectedService = serviceNames[service] || service;

                const message = `Hi A. Swartz DecorHome! I would like to request a quote.\n\n*Name:* ${name}\n*Phone:* ${phone}\n*Service:* ${selectedService}`;
                const whatsappUrl = `https://wa.me/27603993246?text=${encodeURIComponent(message)}`;
                
                // Open WhatsApp in a new tab
                window.open(whatsappUrl, '_blank');

                // Simulate success UI after short delay
                setTimeout(() => {
                    form.classList.add('hidden');
                    successMessage.classList.remove('hidden');
                    successMessage.classList.add('flex');
                    
                    // Reset button in the background for next time
                    submitBtn.disabled = false;
                    btnText.textContent = 'Request Quote';
                    spinner.classList.add('hidden');
                }, 800);
            });

            // Reset form UI back to input state
            resetBtn.addEventListener('click', () => {
                form.reset();
                successMessage.classList.add('hidden');
                successMessage.classList.remove('flex');
                form.classList.remove('hidden');
            });
        });
    </script>
